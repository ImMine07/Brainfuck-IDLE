cmake_minimum_required(VERSION 3.16)
project(bfgui LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

add_executable(bfgui WIN32 bfgui.cpp)

include(FetchContent)
set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OPTION_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
set(OPTION_USE_SYSTEM_LIBPNG OFF CACHE BOOL "" FORCE)
FetchContent_Declare(fltk GIT_REPOSITORY https://github.com/fltk/fltk.git GIT_TAG release-1.3.9)
FetchContent_MakeAvailable(fltk)

target_include_directories(bfgui PRIVATE ${fltk_SOURCE_DIR} ${fltk_BINARY_DIR})
target_link_libraries(bfgui PRIVATE fltk Threads::Threads)

if(MSVC)
  target_compile_options(bfgui PRIVATE /utf-8)
  target_link_options(bfgui PRIVATE /ENTRY:mainCRTStartup)   # WIN32 앱에서도 main() 사용
endif()
